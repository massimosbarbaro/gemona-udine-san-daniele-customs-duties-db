Public Function CheckDate(dDatascadenza As Date) As Boolean
    Dim dDataOggi As Date, fExit As Boolean
    dDataOggi = Now()
    If dDataOggi > dDatascadenza Then
        fExit = False
    Else
        fExit = True
    End If
    CheckDate = fExit
End Function
Public Function Apertura(sApertura As String)
 
        If Right$(sApertura, 1) = ";" Then
            'Devo spezzare la stringa di apertura
            sRic = ";"
            sPosizione = InStr(1, sApertura, sRic, 1)
            sOpen = Left$(sApertura, sPosizione - 1)
            sForm = Mid$(sApertura, sPosizione + 1, (Len(sApertura) - sPosizione - 1))
            
       End If
        

End Function
